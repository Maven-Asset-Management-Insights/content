function zf_CheckWorkEmail() {
  var emailField = document.forms.form["Email"];
  var email = emailField.value.trim();
  var atIndex = email.indexOf('@');
  if (atIndex === -1) return true; // let zf_ValidateAndSubmit's own check handle malformed emails
  var domain = email.substring(atIndex + 1).toLowerCase();
  if (BLOCKED_DOMAINS.indexOf(domain) !== -1) {
    emailField.focus();
    document.getElementById('Email_error').textContent = 'Please use your work email address.';
    document.getElementById('Email_error').style.display = 'block';
    return false;
  }
